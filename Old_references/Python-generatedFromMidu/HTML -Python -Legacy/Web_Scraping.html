Web Scraping con Python

Lección 1: Scraping Básico con `requests` y Regex

La forma más fundamental de hacer scraping es descargar el HTML de una página con requests y luego usar expresiones regulares (regex) para buscar patrones específicos, como el precio o el título.

# 01_basic.py
import requests
import re

url = 'https://www.apple.com/es/shop/buy-mac/macbook-air/'
response = requests.get(url)

if response.status_code == 200:
  print('La petición fue exitosa')
  html = response.text

  # Expresión regular para encontrar el título
  title_pattern = r'(.*?)'
  match = re.search(title_pattern, html)
  if match:
    print(f"El título de la web es: {match.group(1)}")

  # Expresión regular para encontrar el precio
  price_pattern = r'(.*?)'
  match = re.search(price_pattern, html)
  if match:
    print(f"El precio del producto es: {match.group(1)}")
Lección 2: Parseo con `BeautifulSoup`

Las expresiones regulares son frágiles. Una librería como BeautifulSoup nos permite parsear el HTML y navegar por su estructura de una forma mucho más robusta y legible, usando selectores de CSS.

# 02_beautiful.py
from bs4 import BeautifulSoup
import requests

url = 'https://www.apple.com/es/shop/buy-mac/macbook-air/'
headers = {
  'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.124 Safari/537.36'
}
response = requests.get(url, headers=headers)

if response.status_code == 200:
  print('La petición fue exitosa')
  soup = BeautifulSoup(response.text, 'html.parser')

  title_tag = soup.title
  if title_tag:
    print(f"El título de la web es: {title_tag.text}")

  # Encontrar cada producto y obtener su nombre y precio
  products = soup.find_all(class_='rc-productselection-item')
  for product in products:
    name = product.find(class_="list-title").text.strip()
    price = product.find(class_="rc-prices-fullprice").text.strip()
    print(f"Producto: {name}\nPrecio: {price}\n")
Lección 3: Scraper de Múltiples Datos

Podemos crear funciones reutilizables para extraer diferentes tipos de datos de una página, como títulos, enlaces o metadatos (ej. la imagen para compartir en redes sociales).

# 03_wiki_scraper.py
from bs4 import BeautifulSoup
from urllib.parse import urljoin
import requests

def scrape_url(url: str):
  headers = {
    'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.124 Safari/537.36'
  }
  response = requests.get(url, headers=headers)

  if response.status_code == 200:
    print('La petición fue exitosa')
    soup = BeautifulSoup(response.text, 'html.parser')

    # Extraer todos los enlaces y convertirlos a absolutos
    enlaces = [urljoin(url, enlace.get('href')) for enlace in soup.find_all('a')]
    print(f"\nPrimeros 5 enlaces: {enlaces[:5]}")

    # Extraer la imagen de Open Graph (usada para previsualizaciones)
    og_image = soup.find('meta', property='og:image')
    if og_image:
      print(f"\nImagen Open Graph: {og_image['content']}")
    else:
      print('\nNo se encontró la imagen Open Graph')

scrape_url('https://midu.dev')
Lección 4: Creando una Herramienta SEO CLI

Podemos combinar BeautifulSoup con argparse para crear una herramienta de línea de comandos (CLI) que realice comprobaciones básicas de SEO en cualquier URL que le pasemos.

# 04_seo_cli.py
# Para ejecutar: python 04_seo_cli.py https://midu.dev
import requests
import argparse
from bs4 import BeautifulSoup

parser = argparse.ArgumentParser(description="Herramienta SEO para una URL")
parser.add_argument('url', type=str, help='La URL a analizar')
args = parser.parse_args()
url = args.url

headers = {
  'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.124 Safari/537.36'
}
response = requests.get(url, headers=headers)

if response.status_code == 200:
  soup = BeautifulSoup(response.text, 'html.parser')
  print(f"Revisando: {url}")

  # Comprobar título
  titulo_pagina = soup.title.string
  if titulo_pagina:
    print(f"Título: {titulo_pagina}")
    if len(titulo_pagina) < 70:
      print("✅ Longitud de título adecuada")
    else:
      print("⚠️ Título demasiado largo")
  
  # Comprobar H1
  titulos_h1 = soup.find_all('h1')
  if not titulos_h1:
    print("❌ No se encontró H1")
  elif len(titulos_h1) > 1:
    print("⚠️ Hay más de un H1")
  else:
    print("✅ Se encontró un único H1")
Lección 5: Introducción a `Playwright`

Playwright es una herramienta de automatización de navegadores que nos permite controlar un navegador real (Chromium, Firefox, WebKit) mediante código. Es ideal para tests y para interactuar con páginas dinámicas.

# 05_playwright.py
# Es un archivo de test, se ejecuta con: pytest
import re
from playwright.sync_api import Page, expect

def test_has_title(page: Page):
  page.goto("https://playwright.dev/")
  # Espera que el título contenga la palabra "Playwright"
  expect(page).to_have_title(re.compile("Playwright"))

def test_get_started_link(page: Page):
  page.goto("https://playwright.dev/")
  # Hace clic en el enlace "Get started"
  page.get_by_role("link", name="Get started").click()
  # Espera que el H1 con el texto "Installation" sea visible
  expect(page.get_by_role("heading", name="Installation")).to_be_visible()
Lección 6: Scraping con `Playwright`

Cuando una página carga su contenido con JavaScript, requests no es suficiente porque solo descarga el HTML inicial. Playwright ejecuta el JavaScript como un navegador normal, permitiéndonos extraer datos de contenido dinámico.

# 06_playwright_scraping.py
from playwright.sync_api import sync_playwright

url = 'https://midu.dev'

with sync_playwright() as p:
  # headless=False para ver el navegador, slow_mo para ralentizarlo
  browser = p.chromium.launch(headless=True, slow_mo=500)
  page = browser.new_page()
  page.goto(url)

  # Localiza el primer enlace dentro de un 

  first_article_anchor = page.locator('article a').first
  print(f"Texto del primer artículo: {first_article_anchor.text_content()}")
  
  # Hacemos clic y esperamos que la nueva página cargue
  first_article_anchor.click()
  page.wait_for_load_state('load')

  print(f"Nuevo título de la página: {page.title()}")

  browser.close()
