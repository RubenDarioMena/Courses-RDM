Fechas, APIs y Clases en Python

Lección 1: Manejo de Fechas y Horas

Python proporciona el módulo datetime para trabajar con fechas y horas de una manera sencilla y potente.

# 01_dates.py
# Trabajando con fechas y horas en Python

from datetime import datetime, timedelta
import locale

# 1. Obtener la fecha y hora actual
now = datetime.now()
print(f"Fecha y hora actual: {now}")

# 2. Crear una fecha y hora específica
specific_date = datetime(2025, 2, 12, 15, 30, 0)
print(f"Fecha y hora específica: {specific_date}")

# 3. Formatear fechas en español
# Usamos locale para establecer el idioma
locale.setlocale(locale.LC_TIME, 'es_ES.UTF-8')
format_date = now.strftime("%A, %d de %B de %Y - %H:%M:%S")
print(f"Fecha formateada: {format_date}")

# 4. Operaciones con fechas usando timedelta
yesterday = datetime.now() - timedelta(days=1)
print(f"Ayer fue: {yesterday}")

tomorrow = datetime.now() + timedelta(days=1)
print(f"Mañana será: {tomorrow}")

one_hour_after = datetime.now() + timedelta(hours=1)
print(f"Dentro de una hora: {one_hour_after}")

# 5. Obtener componentes individuales de una fecha
year = now.year
month = now.month
day = now.day
print(f"Año: {year}, Mes: {month}, Día: {day}")

# 6. Calcular la diferencia entre dos fechas
date1 = datetime.now()
date2 = datetime(2025, 1, 1)
difference = date2 - date1
print(f"Días hasta año nuevo: {difference.days}")
Lección 2: Peticiones a APIs con `requests`

Para interactuar con servicios web y APIs, la librería requests es el estándar de facto en Python por su simplicidad y poder.

GET, POST, PUT

# 02_requests.py (Parte 1)
import requests

# GET: Obtener datos
print("\nGET:")
api_posts = "https://jsonplaceholder.typicode.com/posts/"
response = requests.get(api_posts)
if response.status_code == 200:
    print(f"GET exitoso! Se obtuvieron {len(response.json())} posts.")
else:
    print(f"Error en GET: {response.status_code}")

# POST: Crear un nuevo recurso
print("\nPOST:")
new_post = {
    "title": "Mi nuevo post",
    "body": "Contenido del post.",
    "userId": 1
}
response = requests.post(api_posts, json=new_post)
if response.status_code == 201: # 201 Created
    print(f"POST exitoso! Nuevo post creado con ID: {response.json()['id']}")
else:
    print(f"Error en POST: {response.status_code}")

# PUT: Actualizar un recurso existente
print("\nPUT:")
updated_post = {
    "id": 1,
    "title": "Título actualizado",
    "body": "Contenido actualizado.",
    "userId": 1
}
response = requests.put(f"{api_posts}1", json=updated_post)
if response.status_code == 200:
    print(f"PUT exitoso! Post actualizado: {response.json()['title']}")
else:
    print(f"Error en PUT: {response.status_code}")

def call_openai_gpt(api_key, prompt):
    url = "https://api.openai.com/v1/chat/completions"
    headers = {
        "Content-Type": "application/json",
        "Authorization": f"Bearer {api_key}"
    }
    data = {
        "model": "gpt-4o-mini",
        "messages": [{"role": "user", "content": prompt}]
    }
    try:
        response = requests.post(url, json=data, headers=headers)
        response.raise_for_status() # Lanza un error para respuestas 4xx/5xx
        return response.json()["choices"][0]["message"]["content"]
    except requests.exceptions.RequestException as e:
        return f"Error en la llamada a OpenAI: {e}"

Lección 3: Clases para Organizar Llamadas a APIs

El encapsulamiento es un pilar de la Programación Orientada a Objetos. Nos permite ocultar la complejidad interna y exponer una interfaz simple. Podemos crear una clase que gestione las llamadas a diferentes APIs de IA, haciendo nuestro código más limpio, reutilizable y fácil de mantener.

# 03_clases.py
import requests

# NOTA: Debes reemplazar con tus propias API keys
OPENAI_KEY = "sk-XXXXXXXX"
DEEPSEEK_API_KEY = "sk-XXXXXXXX"

class AI_API:
    def __init__(self, api_key, url, model):
        if not api_key or api_key == "sk-XXXXXXXX":
            raise ValueError("API key no válida o no proporcionada.")
        self.api_key = api_key
        self.url = url
        self.model = model

    def call(self, prompt):
        headers = {
            "Content-Type": "application/json",
            "Authorization": f"Bearer {self.api_key}"
        }
        data = {
            "model": self.model,
            "messages": [{"role": "user", "content": prompt}]
        }

        try:
            response = requests.post(self.url, json=data, headers=headers)
            response.raise_for_status() # Lanza error si la petición falla
            return response.json()["choices"][0]["message"]["content"]
        except requests.exceptions.RequestException as e:
            print(f"Error en la solicitud a {self.url}: {e}")
            return None
        except (KeyError, IndexError) as e:
            print(f"Error procesando la respuesta de la API: {e}")
            return None

# --- Uso de la clase --- 

poema_prompt = "Escribe un breve poema sobre la programación"

try:
    print("\nLlamando a OpenAI con la clase:")
    openai_api = AI_API(OPENAI_KEY, "https://api.openai.com/v1/chat/completions", "gpt-4o-mini")
    respuesta_openai = openai_api.call(poema_prompt)
    if respuesta_openai:
        print(respuesta_openai)
except ValueError as e:
    print(e)

try:
    print("\nLlamando a DeepSeek con la clase:")
    deepseek_api = AI_API(DEEPSEEK_API_KEY, "https://api.deepseek.com/chat/completions", "deepseek-chat")
    respuesta_deepseek = deepseek_api.call(poema_prompt)
    if respuesta_deepseek:
        print(respuesta_deepseek)
except ValueError as e:
    print(e)
