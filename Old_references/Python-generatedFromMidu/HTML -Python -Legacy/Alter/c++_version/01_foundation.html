Lecciones Fundamentales

La función std::cout es fundamental en C++. Permite mostrar información en la consola, una herramienta que te acompañará durante toda tu carrera como programador. Se utiliza para depurar código, mostrar resultados y comunicarte con el usuario. Puedes imprimir texto (cadenas), números y los valores de las variables. A diferencia de Python, C++ requiere incluir la librería <iostream> y usar el operador << para enviar datos al flujo de salida.

/*
 * 01 - cout (salida de datos)
 * std::cout es el objeto que nos permite imprimir en consola
 * Sirve para mostrar información en consola y te va a acompañar
 * TODA TU VIDA. Desde hoy hasta el fin de los tiempos
 */

#include   // Necesario para cout
#include     // Necesario para std::string

// Función para limpiar pantalla (multiplataforma)
void clearScreen() {
    #ifdef _WIN32
        system("cls");
    #else
        system("clear");
    #endif
}

int main() {
    clearScreen();
    
    // Este es un ejemplo básico de cómo imprimir un texto en consola
    std::cout << "¡Hola, Twitch!" << std::endl;
    
    // También puedes usar comillas dobles para imprimir texto
    std::cout << "Esto funciona con comillas dobles" << std::endl;
    
    // Puedes imprimir múltiples elementos en una sola línea
    std::cout << "C++" << " " << "es" << " " << "genial" << std::endl;
    
    // Usando std::endl vs \n
    std::cout << "Esto se imprime" << std::endl; // std::endl hace flush del buffer
    std::cout << "en líneas separadas\n";       // \n es más eficiente
    
    // También se pueden imprimir números directamente
    std::cout << 42 << std::endl;
    
    // Ejemplo de cómo imprimir comillas dentro de strings
    std::cout << "Esto es una \"pulgada\" dentro de un string" << std::endl;
    
    // Usando caracteres de escape
    std::cout << "Línea 1\nLínea 2\tTabulación" << std::endl;
    
    return 0;  // Indica que el programa terminó exitosamente
}

C++ es un lenguaje de tipado estático y fuerte. Esto significa que debes declarar el tipo de cada variable antes de usarla, y el tipo no puede cambiar durante la ejecución. Los tipos fundamentales incluyen int para números enteros, double y float para números decimales, char para caracteres individuales, bool para valores verdaderos/falsos, y std::string para cadenas de texto. La función typeid() puede ayudarte a identificar el tipo de una variable.

/*
 * 02 - Tipos de datos fundamentales
 * C++ tiene varios tipos de datos primitivos
 * int, float, double, char, bool, string...
 */

#include 
#include 
#include   // Para typeid()

int main() {
    // Números enteros
    int numero_entero = 10;
    int cero = 0;
    int negativo = -5;
    long long numero_grande = 7238424723784278934LL;  // Para números muy grandes
    
    std::cout << "int:" << std::endl;
    std::cout << "Valor: " << numero_entero << ", Tipo: " << typeid(numero_entero).name() << std::endl;
    std::cout << "Valor: " << cero << ", Tipo: " << typeid(cero).name() << std::endl;
    std::cout << "Valor: " << negativo << ", Tipo: " << typeid(negativo).name() << std::endl;
    
    // Números decimales
    float decimal_simple = 3.14f;      // f indica que es float
    double decimal_doble = 3.14159;    // double tiene más precisión
    double notacion_cientifica = 1e3;  // Equivalente a 1000.0
    
    std::cout << "\nfloat y double:" << std::endl;
    std::cout << "Float: " << decimal_simple << ", Tipo: " << typeid(decimal_simple).name() << std::endl;
    std::cout << "Double: " << decimal_doble << ", Tipo: " << typeid(decimal_doble).name() << std::endl;
    std::cout << "Científica: " << notacion_cientifica << ", Tipo: " << typeid(notacion_cientifica).name() << std::endl;
    
    // Caracteres
    char letra = 'A';
    char simbolo = '@';
    
    std::cout << "\nchar:" << std::endl;
    std::cout << "Valor: " << letra << ", Tipo: " << typeid(letra).name() << std::endl;
    std::cout << "Valor: " << simbolo << ", Tipo: " << typeid(simbolo).name() << std::endl;
    
    // Cadenas de texto
    std::string texto = "Hola mundo";
    std::string vacio = "";
    std::string numero_como_texto = "123";
    
    std::cout << "\nstd::string:" << std::endl;
    std::cout << "Valor: \"" << texto << "\", Tipo: " << typeid(texto).name() << std::endl;
    std::cout << "Vacío: \"" << vacio << "\", Tipo: " << typeid(vacio).name() << std::endl;
    
    // Valores booleanos
    bool verdadero = true;
    bool falso = false;
    bool comparacion = (1 < 2);
    
    std::cout << "\nbool:" << std::endl;
    std::cout << "Verdadero: " << std::boolalpha << verdadero << ", Tipo: " << typeid(verdadero).name() << std::endl;
    std::cout << "Falso: " << falso << ", Tipo: " << typeid(falso).name() << std::endl;
    std::cout << "Comparación: " << comparacion << ", Tipo: " << typeid(comparacion).name() << std::endl;
    
    return 0;
}

El "casting" en C++ es el proceso de convertir un valor de un tipo a otro. C++ ofrece varios tipos de casting: implícito (automático y seguro), explícito usando operadores de casting como static_cast<>, y casting estilo C (menos seguro). Es importante entender cuándo las conversiones son seguras y cuándo pueden perder información. Por ejemplo, convertir un double a int trunca la parte decimal.

/*
 * 03 - Conversión de tipos (Casting)
 * Transformar un tipo de un valor a otro en C++
 */

#include 
#include 
#include   // Para std::round

int main() {
    std::cout << "Conversión de tipos en C++" << std::endl;
    
    // Conversión de string a número (usando stoi, stof, stod)
    std::string numero_texto = "100";
    int numero_convertido = std::stoi(numero_texto);  // string to int
    std::cout << "2 + " << numero_texto << " = " << 2 + numero_convertido << std::endl;
    
    // Conversión de número a string (usando to_string)
    int numero = 2;
    std::string resultado = numero_texto + std::to_string(numero);
    std::cout << "\"100\" + \"2\" = \"" << resultado << "\"" << std::endl;
    
    // Conversión de string a double
    std::string pi_texto = "3.1416";
    double pi = std::stod(pi_texto);  // string to double
    std::cout << "Tipo de pi: " << typeid(pi).name() << ", Valor: " << pi << std::endl;
    
    // Conversión de double a int (trunca la parte decimal)
    int pi_entero = static_cast(pi);  // Casting explícito y seguro
    std::cout << "Pi como entero (truncado): " << pi_entero << std::endl;
    
    // Conversiones a bool
    bool numero_bool = static_cast(3);    // Cualquier número != 0 es true
    bool cero_bool = static_cast(0);      // 0 es false
    bool negativo_bool = static_cast(-1); // Números negativos son true
    
    std::cout << "\nConversiones a bool:" << std::endl;
    std::cout << "bool(3): " << std::boolalpha << numero_bool << std::endl;
    std::cout << "bool(0): " << cero_bool << std::endl;
    std::cout << "bool(-1): " << negativo_bool << std::endl;
    
    // Conversiones de string a bool (más complejo en C++)
    std::string texto_vacio = "";
    std::string texto_con_contenido = " ";
    std::cout << "\nStrings como bool (usando .empty()):" << std::endl;
    std::cout << "String vacío es: " << (texto_vacio.empty() ? "true" : "false") << std::endl;
    std::cout << "String con espacio es: " << (texto_con_contenido.empty() ? "true" : "false") << std::endl;
    
    // Redondeo
    double decimal = 2.51;
    int redondeado = static_cast(std::round(decimal));
    std::cout << "\nRedondeo de " << decimal << ": " << redondeado << std::endl;
    
    // Ejemplo de conversión que fallaría (comentado para evitar error)
    // int error = std::stoi("Hola mundo");  // ❌ Esto lanzaría std::invalid_argument
    
    return 0;
}

Las variables en C++ deben declararse con un tipo específico antes de usarse. C++ es de tipado estático (el tipo se define en tiempo de compilación) y tipado fuerte (no permite conversiones automáticas peligrosas). Es importante seguir las convenciones de nomenclatura como snake_case o camelCase. C++ también permite declarar constantes usando const y variables automáticas usando auto (el compilador deduce el tipo).

/*
 * 04 - Variables y declaraciones
 * Las variables sirven para guardar datos en memoria.
 * C++ es un lenguaje de tipado estático y de tipado fuerte.
 */

#include 
#include 

int main() {
    // Declaración e inicialización de variables
    std::string my_name = "midudev";
    std::cout << my_name << std::endl;
    
    int age = 32;
    std::cout << age << std::endl;
    
    // Reasignar un nuevo valor a una variable existente
    age = 39;
    std::cout << age << std::endl;
    
    // Tipado estático: el tipo se determina en tiempo de compilación
    std::string name = "midudev";
    std::cout << "Tipo de name: " << typeid(name).name() << std::endl;
    
    // ❌ Esto NO es posible en C++ (a diferencia de Python)
    // name = 32;  // Error de compilación: no se puede asignar int a string
    
    // Tipado fuerte: C++ no realiza conversiones automáticas peligrosas
    // std::cout << 10 + "2" << std::endl;  // ❌ Error de compilación
    
    // Concatenación de strings en C++ (equivalente a f-strings de Python)
    std::cout << "Hola " << my_name << ", tengo " << age + 5 << " años" << std::endl;
    
    // Declaración múltiple (no recomendada)
    std::string nombre = "midudev";
    int edad = 32;
    std::string ciudad = "Bogotá";
    
    // Convenciones de nombres de variables
    std::string mi_nombre_de_variable = "ok";     // snake_case
    std::string nombre_simple = "ok";
    
    std::string miNombreDeVariable = "también_ok"; // camelCase (común en C++)
    std::string MiNombreDeVariable = "PascalCase"; // Para clases/tipos
    std::string minombredevariable = "no-recomendado"; // todo junto
    
    std::string mi_nombre_de_variable_123 = "ok";
    
    // Constantes (equivalente a variables en MAYÚSCULAS de Python)
    const double MI_CONSTANTE = 3.14159;
    // MI_CONSTANTE = 2.71;  // ❌ Error: no se puede modificar una constante
    
    // Deducción automática de tipos (C++11 y posterior)
    auto numero_automatico = 42;        // El compilador deduce que es int
    auto decimal_automatico = 3.14;     // El compilador deduce que es double
    auto texto_automatico = "Hola";     // El compilador deduce que es const char*
    
    std::cout << "\nTipos deducidos automáticamente:" << std::endl;
    std::cout << "auto numero: " << typeid(numero_automatico).name() << std::endl;
    std::cout << "auto decimal: " << typeid(decimal_automatico).name() << std::endl;
    std::cout << "auto texto: " << typeid(texto_automatico).name() << std::endl;
    
    // Inicialización uniforme (C++11)
    int valor{42};
    std::string mensaje{"Hola mundo"};
    
    return 0;
}

La función std::cin permite obtener datos del usuario desde la consola. A diferencia de Python's input(), cin puede leer diferentes tipos de datos directamente sin necesidad de conversión manual. Sin embargo, es importante manejar los espacios en blanco y usar getline() para leer líneas completas con espacios. También es crucial validar la entrada para evitar errores.

/*
 * 05 - Entrada de usuario (cin)
 * std::cin permite obtener datos del usuario a través de la consola.
 */

#include 
#include 
#include   // Para stringstream

int main() {
    // Para obtener datos del usuario se usa std::cin
    std::string nombre;
    std::cout << "Hola, ¿cómo te llamas?\n";
    std::cin >> nombre;  // Lee hasta el primer espacio
    std::cout << "Hola " << nombre << ", encantado de conocerte" << std::endl;
    
    // Para leer números, cin puede convertir automáticamente
    int age;
    std::cout << "¿Cuántos años tienes?\n";
    std::cin >> age;
    std::cout << "Tienes " << age << " años" << std::endl;
    
    // Limpiar el buffer antes de usar getline
    std::cin.ignore();
    
    // Para leer líneas completas (incluyendo espacios), usar getline
    std::string nombre_completo;
    std::cout << "¿Cuál es tu nombre completo?\n";
    std::getline(std::cin, nombre_completo);
    std::cout << "Tu nombre completo es: " << nombre_completo << std::endl;
    
    // Obtener múltiples valores en una línea
    std::string pais, ciudad;
    std::cout << "¿En qué país y ciudad vives? (separados por espacio)\n";
    std::cin >> pais >> ciudad;
    std::cout << "Vives en " << pais << ", " << ciudad << std::endl;
    
    // Validación básica de entrada
    int numero;
    std::cout << "Ingresa un número entero:\n";
    while (!(std::cin >> numero)) {
        std::cout << "Error: Por favor ingresa un número válido:\n";
        std::cin.clear();  // Limpiar el estado de error
        std::cin.ignore(10000, '\n');  // Descartar la entrada inválida
    }
    std::cout << "El número ingresado es: " << numero << std::endl;
    
    return 0;
}

A diferencia de Python, C++ es un lenguaje compilado. Esto significa que debes convertir tu código fuente en un archivo ejecutable antes de poder ejecutarlo. El proceso incluye: preprocesamiento (manejo de #include y #define), compilación (conversión a código objeto), enlazado (linking de librerías), y finalmente la ejecución. Los compiladores más comunes son GCC, Clang y MSVC.

# Compilación básica con g++
g++ -o programa programa.cpp

# Compilación con flags de optimización y debug
g++ -std=c++17 -Wall -Wextra -g -O2 -o programa programa.cpp

# Ejecución del programa
./programa

# Compilación paso a paso (opcional, para entender el proceso)
g++ -E programa.cpp > programa.i    # Preprocesamiento
g++ -S programa.i                   # Compilación a ensamblador
g++ -c programa.s                   # Ensamblado a código objeto
g++ programa.o -o programa          # Enlazado final
/*
 * 06 - Estructura básica de un programa C++
 * Todo programa C++ necesita estas partes fundamentales
 */

// Directivas de preprocesador (incluir librerías)
#include   // Para entrada/salida estándar
#include     // Para manejo de strings

// Declaración del espacio de nombres (opcional pero común)

// Función principal (punto de entrada del programa)
int main() {

}
